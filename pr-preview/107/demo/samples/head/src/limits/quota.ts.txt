import type { Options } from './options';

/** What a client has left in the current window. */
export function remaining(client: string, usage: Map<string, number>, options: Options) {
  const used = usage.get(client) ?? 0;
  const spare = options.perMinute - used;
  return Math.max(0, spare);
}

/** When a rejected client may try again, in seconds. */
export function retryAfter(windowStart: number, now: number) {
  return Math.max(0, 60 - Math.floor((now - windowStart) / 1000));
}
