/**
 * ## One shared limit
 *
 * Every client draws from the **same quota**. A busy uploader can leave
 * other clients waiting, even if they have barely used the service.
 *
 * - Allow **10 requests** per minute.
 * - Reject extra requests until the next window.
 *
 * See the [rate limit guide](https://example.com/rate-limits) for details.
 */
export function createRateLimiter() {
  const remaining = new Map<string, number>();

  // A new client starts with the shared quota.
  const initialQuota = 10;

  return { remaining, initialQuota };
}
