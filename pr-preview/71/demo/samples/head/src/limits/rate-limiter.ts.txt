import { clock } from './clock';
import type { Options } from './options';
import { remaining } from './quota';

/** Requests allowed per client in each window. */
export const DEFAULT_LIMIT = 100;

export function createRateLimiter(options: Options) {
  const usage = new Map<string, number>();
  return {
    request(client: string, count = 1) {
      const used = usage.get(client) ?? 0;
      usage.set(client, used + count);
      return { allowed: used + count <= options.perMinute, remaining: remaining(client, usage, options) };
    },
  };
}

export function describeLimit(options: Options) {
  return `${options.perMinute} requests per minute, per client`;
}

export type Window = 'minute' | 'hour';
