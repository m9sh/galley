import { clock } from './clock';
import type { Options } from './options';

/** Requests allowed per client in each window. */
export const DEFAULT_LIMIT = 100;

export function createRateLimiter(options: Options) {
  const usage = new Map<string, number>();
  return {
    request(client: string) {
      const used = usage.get(client) ?? 0;
      usage.set(client, used + 1);
      return { allowed: used < options.perMinute, remaining: remaining(client, usage, options) };
    },
  };
}

/** What a client has left in the current window. */
export function remaining(client: string, usage: Map<string, number>, options: Options) {
  const used = usage.get(client) ?? 0;
  const spare = options.perMinute - used;
  return Math.max(0, spare);
}

export function resetAll(usage: Map<string, number>) {
  usage.clear();
  clock.restart();
}

export function describeLimit(options: Options) {
  return `${options.perMinute} requests per minute`;
}
