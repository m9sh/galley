import { beforeEach, describe, expect, it } from 'vitest';
import { createRateLimiter } from './rate-limiter';

describe('Rate limits for batch uploads', () => {
  let limiter: ReturnType<typeof createRateLimiter>;

  beforeEach(() => {
    limiter = createRateLimiter({ perMinute: 10 });
  });

  it('accepts requests within the limit', () => {
    const response = limiter.request('client-a');

    expect(response.allowed).toBe(true);
    expect(response.remaining).toBe(9);
  });

  it('rejects requests after the limit is reached', () => {
    for (let i = 0; i < 10; i++) {
      limiter.request('client-a');
    }

    expect(limiter.request('client-a').allowed).toBe(false);
  });

  it('shares one limit across every client', () => {
    limiter.request('client-a');

    expect(limiter.request('client-b').remaining).toBe(8);
  });
  it('combines usage from unrelated clients', () => {
    limiter.request('client-a');
    limiter.request('client-c');
    expect(limiter.request('client-b').remaining).toBe(7);
  });
});
