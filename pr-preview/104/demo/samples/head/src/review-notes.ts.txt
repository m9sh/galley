/**
 * ## A fair limit for each client
 *
 * Every client gets its **own quota**. A busy uploader can keep working
 * without making other clients wait.
 *
 * - Allow **100 requests** per minute, per client.
 * - Return a `retryAfter` value when the quota runs out.
 *
 * See the [rate limit guide](https://example.com/rate-limits) for details.
 */
export function createRateLimiter() {
  const remaining = new Map<string, number>();

  // A new client starts with a full, independent quota.
  const initialQuota = 100;

  return { remaining, initialQuota };
}
